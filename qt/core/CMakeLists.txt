# lyrics_card.js goes into the library as bytes: page_scripts.cpp includes generated/lyrics_card_js.inc, which defines
# kLyricsCardJs. Written when CMake runs, and CMake runs again whenever the script changes (CMAKE_CONFIGURE_DEPENDS);
# file(CONFIGURE) leaves the file alone when its content is the same.
set(XGS_LYRICS_CARD_JS ${CMAKE_CURRENT_SOURCE_DIR}/lyrics/google/lyrics_card.js)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${XGS_LYRICS_CARD_JS})
file(READ ${XGS_LYRICS_CARD_JS} XGS_LYRICS_CARD_HEX HEX)
string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," XGS_LYRICS_CARD_BYTES "${XGS_LYRICS_CARD_HEX}")
string(REPEAT "0x[0-9a-f][0-9a-f]," 16 XGS_BYTE_ROW)
string(REGEX REPLACE "(${XGS_BYTE_ROW})" "\\1\n" XGS_LYRICS_CARD_BYTES "${XGS_LYRICS_CARD_BYTES}")
file(CONFIGURE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/generated/lyrics_card_js.inc
    CONTENT "// Generated by qt/core/CMakeLists.txt from lyrics/google/lyrics_card.js. Do not edit.\nconstexpr unsigned char kLyricsCardJs[] = {\n@XGS_LYRICS_CARD_BYTES@\n};\n"
    @ONLY)
target_include_directories(xgscore PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_link_libraries(xgscore PUBLIC Qt6::Core Qt6::Network)
xgs_target_defaults(xgscore)
